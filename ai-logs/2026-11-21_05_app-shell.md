> # Test-Driven Development (TDD)
> 
> ## Overview
> 
> Write the test first. Watch it fail. Write minimal code to pass.
> 
> **Core principle:** If you didn't watch the test fail, you don't know if it tests the right thing.
> 
> **Violating the letter of the rules is violating the spirit of the rules.**
> 
> ## When to Use
> 
> **Always:**
> - New features
> - Bug fixes
> - Refactoring
> - Behavior changes
> 
> **Exceptions (ask your human partner):**
> - Throwaway prototypes
> - Generated code
> - Configuration files
> 
> Thinking "skip TDD just this once"? Stop. That's rationalization.
> 
> ## The Iron Law
> 
> ```
> NO PRODUCTION CODE WITHOUT A FAILING TEST FIRST
> ```
> 
> Write code before the test? Delete it. Start over.
> 
> **No exceptions:**
> - Don't keep it as "reference"
> - Don't "adapt" it while writing tests
> - Don't look at it
> - Delete means delete
> 
> Implement fresh from tests. Period.
> 
> ## Red-Green-Refactor
> 
> ```dot
> digraph tdd_cycle {
>     rankdir=LR;
>     red [label="RED\nWrite failing test", shape=box, style=filled, fillcolor="#ffcccc"];
>     verify_red [label="Verify fails\ncorrectly", shape=diamond];
>     green [label="GREEN\nMinimal code", shape=box, style=filled, fillcolor="#ccffcc"];
>     verify_green [label="Verify passes\nAll green", shape=diamond];
>     refactor [label="REFACTOR\nClean up", shape=box, style=filled, fillcolor="#ccccff"];
>     next [label="Next", shape=ellipse];
> 
>     red -> verify_red;
>     verify_red -> green [label="yes"];
>     verify_red -> red [label="wrong\nfailure"];
>     green -> verify_green;
>     verify_green -> refactor [label="yes"];
>     verify_green -> green [label="no"];
>     refactor -> verify_green [label="stay\ngreen"];
>     verify_green -> next;
>     next -> red;
> }
> ```
> 
> ### RED - Write Failing Test
> 
> Write one minimal test showing what should happen.
> 
> <Good>
> ```typescript
> test('retries failed operations 3 times', async () => {
>   let attempts = 0;
>   const operation = () => {
>     attempts++;
>     if (attempts < 3) throw new Error('fail');
>     return 'success';
>   };
> 
>   const result = await retryOperation(operation);
> 
>   expect(result).toBe('success');
>   expect(attempts).toBe(3);
> });
> ```
> Clear name, tests real behavior, one thing
> </Good>
> 
> <Bad>
> ```typescript
> test('retry works', async () => {
>   const mock = jest.fn()
>     .mockRejectedValueOnce(new Error())
>     .mockRejectedValueOnce(new Error())
>     .mockResolvedValueOnce('success');
>   await retryOperation(mock);
>   expect(mock).toHaveBeenCalledTimes(3);
> });
> ```
> Vague name, tests mock not code
> </Bad>
> 
> **Requirements:**
> - One behavior
> - Clear name
> - Real code (no mocks unless unavoidable)
>

import { useCallback, useMemo } from "react";
import { useNavigate, useSearchParams } from "react-router";
import { buildStoreSearch, mergeStoreParams, parseStoreParams, type StoreParams } from "./storeParams.ts";

export interface SetParamsOptions {
  /** Replace the current history entry instead of adding one (use it for things like typing a price). */
  replace?: boolean;
}

/**
 * The Store filters, read from and written to the URL. The URL is the single source of truth:
 * there is no copy in React state, so reload, Back/Forward and shared links always agree with the screen.
 *
 *   const { params, setParams, resetParams } = useStoreParams();
 *   setParams({ platform: ["steam"] });   // -> /store?platform=steam (page goes back to 1)
 *
 * Reading uses useSearchParams. Writing goes through navigate() with a string we build ourselves
 * (buildStoreSearch), because setSearchParams would turn the commas into %2C.
 */
export function useStoreParams() {
  const [searchParams] = useSearchParams();

  // URLSearchParams is a new object on every render; its text is stable, so memoize on that.
  const search = searchParams.toString();
  const params = useMemo(() => parseStoreParams(new URLSearchParams(search)), [search]);

  const navigate = useNavigate();

  const setParams = useCallback(
    (patch: Partial<StoreParams>, options: SetParamsOptions = {}) => {
      // Only `search` is given, so the path stays the same (we are on /store).
      navigate({ search: buildStoreSearch(mergeStoreParams(params, patch)) }, { replace: options.replace });
    },
    [navigate, params],
  );

  const resetParams = useCallback(() => navigate({ search: "" }), [navigate]);

  return { params, setParams, resetParams };
}

import type { ComponentProps, MouseEvent } from "react";
import { cx } from "../lib/cx.ts";
import styles from "./Button.module.css";

export interface ButtonProps extends ComponentProps<"button"> {
  /** primary = the main action (once per view area), secondary = outlined, ghost = text-like. */
  variant?: "primary" | "secondary" | "ghost";
  /** Shows a spinner instead of the label, keeps the size and ignores clicks. */
  loading?: boolean;
}

/**
 * The one button of the app (DESIGN.md 4.1). `disabled` is the normal HTML attribute.
 * While `loading` the button is NOT disabled (it would lose keyboard focus); it is marked
 * aria-busy and its clicks are swallowed instead.
 */
export function Button({
  variant = "primary",
  loading = false,
  type = "button", // a <button> inside a <form> submits by default; make that opt-in
  className,
  children,
  onClick,
  ...rest
}: ButtonProps) {
  function handleClick(event: MouseEvent<HTMLButtonElement>) {
    if (loading) {
      event.preventDefault(); // also stops a submit button from submitting
      return;
    }
    onClick?.(event);
  }

  return (
    <button
      {...rest}
      type={type}
      className={cx(styles.button, styles[variant], loading && styles.loading, className)}
      aria-busy={loading || undefined}
      onClick={handleClick}
    >
      <span className={styles.label}>{children}</span>
      {loading && <span className={styles.spinner} aria-hidden="true" />}
    </button>
  );
}

import { useEffect, type RefObject } from "react";
import { isTypingTarget } from "./isTypingTarget.ts";

/**
 * Pressing "/" anywhere on the page focuses the given input (the same shortcut GitHub and YouTube use).
 * It does nothing while the user is already typing in a field (then "/" is just a character),
 * and it ignores combinations like Ctrl+/ that belong to the browser.
 */
export function useSlashShortcut(inputRef: RefObject<HTMLInputElement | null>) {
  useEffect(() => {
    function onKeyDown(event: KeyboardEvent) {
      if (event.key !== "/" || event.ctrlKey || event.metaKey || event.altKey) return;
      if (event.defaultPrevented || isTypingTarget(event.target)) return;
      event.preventDefault(); // otherwise the "/" would be typed into the field we just focused
      inputRef.current?.focus();
    }
    document.addEventListener("keydown", onKeyDown);
    return () => document.removeEventListener("keydown", onKeyDown);
  }, [inputRef]);
}
